**Goal:** Add a minimal BubbleTea TUI (form + live run + summary) to v2pingo without changing existing CLI behavior.

**Architecture:** New isolated `pkg/tui` package (collector, model, run entry). `cmd/v2pingo/main.go` gains a `-tui` flag (default true on an interactive terminal, false when piped) that branches to `tui.Run`; the CLI path is byte-for-byte unchanged. TUI forwards each `tester.RunPool` result into the BubbleTea loop via `program.Send`.

**Tech Stack:** Go 1.27, `charmbracelet/bubbletea`, `charmbracelet/bubbles` (textinput, spinner, progress, viewport), `charmbracelet/lipgloss`, existing `pkg/tester`, `pkg/subscription`, `pkg/parser`, `v2pingo/pkg/types`.

**Spec:** `docs/superpowers/specs/2026-09-27-tui-design.md`

## Global Constraints

- Windows single-`.exe` build must keep working: `go build -o v2pingo.exe ./cmd/v2pingo`.
- CLI regression path (`-tui=false`) behaves exactly like today.
- Saved results rule (verbatim from spec): only `Success == true` with `200 <= StatusCode < 300` and non-empty `RawLink`.
- Workers range 1..200; timeout parsed with `time.ParseDuration`; at least one subscription required.

package tui

import (
	"sync"

	"v2pingo/pkg/types"
)

// Collector is the TUI-side equivalent of reporter.Reporter accounting.
// It is safe for concurrent use from tester.RunPool worker callbacks.
type Collector struct {
	mu      sync.Mutex
	tested  int
	success int
	failed  int
	links   []string
}

// OnResult records one TestResult. Only Success results with a 2xx status
// and a non-empty RawLink are kept for the results file.
func (c *Collector) OnResult(res types.TestResult) {
	c.mu.Lock()
	defer c.mu.Unlock()
	c.tested++
	if res.Success {
		c.success++
		if res.StatusCode >= 200 && res.StatusCode < 300 && res.Config.RawLink != "" {
			c.links = append(c.links, res.Config.RawLink)
		}
		return
	}
	c.failed++
}

// Snapshot returns a copy of the current counters and saved links.
func (c *Collector) Snapshot() (tested, success, failed int, links []string) {
	c.mu.Lock()
	defer c.mu.Unlock()
	out := make([]string, len(c.links))
	copy(out, c.links)
	return c.tested, c.success, c.failed, out
}
```

package tui

import (
	"strconv"
	"strings"
	"time"
)

// FormConfig is the validated TUI form output handed to the runner.
type FormConfig struct {
	Subs      []string
	TargetURL string
	Workers   int
	Timeout   time.Duration
}

// ValidateForm checks raw form strings. It returns the config and "" on
// success, or an empty config plus a human-readable error message.
func ValidateForm(subs []string, extraSub, target, workersStr, timeoutStr string) (FormConfig, string) {
	all := make([]string, 0, len(subs)+1)
	for _, s := range subs {
		if trimmed := strings.TrimSpace(s); trimmed != "" {
			all = append(all, trimmed)
		}
	}
	if trimmed := strings.TrimSpace(extraSub); trimmed != "" {
		all = append(all, trimmed)
	}
	if len(all) == 0 {
		return FormConfig{}, "at least one subscription URL is required"
	}
	if strings.TrimSpace(target) == "" {
		return FormConfig{}, "target URL must not be empty"
	}
	w, err := strconv.Atoi(strings.TrimSpace(workersStr))
	if err != nil || w < 1 || w > 200 {
		return FormConfig{}, "workers must be a number between 1 and 200"
	}
	d, err := time.ParseDuration(strings.TrimSpace(timeoutStr))
	if err != nil || d <= 0 {
		return FormConfig{}, "timeout must be a valid duration like 15s"
	}
	return FormConfig{Subs: all, TargetURL: strings.TrimSpace(target), Workers: w, Timeout: d}, ""
}

package tui

import (
	"fmt"
	"strings"
	"time"

	"github.com/charmbracelet/bubbles/spinner"
	"github.com/charmbracelet/bubbles/textinput"
	"github.com/charmbracelet/bubbles/viewport"
	tea "github.com/charmbracelet/bubbletea"
	"v2pingo/pkg/types"
}

// Screen is the visible TUI page.
type Screen int

const (
	ScreenForm Screen = iota
	ScreenRunning
	ScreenDone
)

// ResultMsg carries one finished proxy test into the BubbleTea loop.
type ResultMsg struct{ Res types.TestResult }

// DoneMsg ends the run with final counters and the saved-file path.
type DoneMsg struct {
	Tested    int
	Success   int
	Failed    int
	Links     []string
	SavedFile string
	Elapsed   time.Duration
}

// StartTestsMsg triggers the background fetch+test goroutine.
type StartTestsMsg struct{}

// Model is the single BubbleTea model for form, running, and done screens.
type Model struct {
	screen    Screen
	inputs    []textinput.Model
	focus     int
	formErr   string
	subs      []string
	total     int
	tested    int
	succeeded int
	failed    int
	links     []string
	recent    []string
	savedFile string
	elapsed   time.Duration
	start     time.Time
	spinner   spinner.Model
	viewport  viewport.Model
}

// NewModel builds the form screen. subs are the preloaded subscription lines
// (from sub-input.txt); target/workers/timeout are the CLI flag defaults.
func NewModel(subs []string, target string, workers int, timeout time.Duration) Model {
	m := Model{screen: ScreenForm, subs: subs}
	fields := []struct {
		placeholder string
		value       string
	}{
		{"extra sub URL (optional)", ""},
		{"target URL", target},
		{"workers 1-200", fmt.Sprintf("%d", workers)},
		{"timeout, e.g. 15s", timeout.String()},
	}
	for i, f := range fields {
		ti := textinput.New()
		ti.Placeholder = f.placeholder
		ti.SetValue(f.value)
		if i == 0 {
			ti.Focus()
		}
		m.inputs = append(m.inputs, ti)
	}
	m.spinner = spinner.New()
	m.viewport = viewport.New(60, 10)
	return m
}

// Update routes messages and keys to the active screen.
func (m Model) Update(msg tea.Msg) (tea.Model, tea.Cmd) {
	switch msg := msg.(type) {
	case tea.KeyMsg:
		switch msg.Type {
		case tea.KeyCtrlC:
			return m, tea.Quit
		case tea.KeyTab:
			if m.screen == ScreenForm {
				m.inputs[m.focus].Blur()
				m.focus = (m.focus + 1) % len(m.inputs)
				m.inputs[m.focus].Focus()
			}
			return m, nil
		case tea.KeyEnter:
			if m.screen == ScreenForm {
				cfg, errMsg := ValidateForm(m.subs, m.inputs[0].Value(), m.inputs[1].Value(), m.inputs[2].Value(), m.inputs[3].Value())
				if errMsg != "" {
					m.formErr = errMsg
					return m, nil
				}
				m.formErr = ""
				m.screen = ScreenRunning
				m.start = time.Now()
				_ = cfg
				return m, func() tea.Msg { return StartTestsMsg{} }
			}
			if m.screen == ScreenDone {
				m.screen = ScreenForm
				return m, nil
			}
			return m, nil
		}
		var cmd tea.Cmd
		if m.screen == ScreenForm {
			m.inputs[m.focus], cmd = m.inputs[m.focus].Update(msg)
		}
		return m, cmd
	case ResultMsg:
		m.tested++
		if msg.Res.Success {
			m.succeeded++
			if msg.Res.StatusCode >= 200 && msg.Res.StatusCode < 300 && msg.Res.Config.RawLink != "" {
				m.links = append(m.links, msg.Res.Config.RawLink)
				line := fmt.Sprintf("[ok] %s", strings.TrimSpace(msg.Res.Config.RawLink))
				m.recent = append(m.recent, line)
				if len(m.recent) > 10 {
					m.recent = m.recent[len(m.recent)-10:]
				}
				m.viewport.SetContent(strings.Join(m.recent, "\n"))
			}
			return m, nil
		}
		m.failed++
		return m, nil
	case DoneMsg:
		m.screen = ScreenDone
		m.tested, m.succeeded, m.failed = msg.Tested, msg.Success, msg.Failed
		m.links, m.savedFile, m.elapsed = msg.Links, msg.SavedFile, msg.Elapsed
		return m, nil
	}
	return m, nil
}

// View renders the active screen. Full styling is a follow-up; this keeps
// the state machine testable and the screens readable.
func (m Model) View() string {
	switch m.screen {
	case ScreenRunning:
		return fmt.Sprintf("Testing %d/%d  ok=%d fail=%d\n%s", m.tested, m.total, m.succeeded, m.failed, m.viewport.View())
	case ScreenDone:
		return fmt.Sprintf("Done: tested=%d success=%d failed=%d in %s\nSaved: %s\n(Enter=form q=quit)", m.tested, m.succeeded, m.failed, m.elapsed.Round(time.Second), m.savedFile)
	default:
		var b strings.Builder
		b.WriteString("v2pingo — Tab moves, Enter starts, Ctrl+C quits\n")
		for _, in := range m.inputs {
			b.WriteString(in.View() + "\n")
		}
		if m.formErr != "" {
			b.WriteString("error: " + m.formErr + "\n")
		}
		return b.String()
	}
}

// Init starts the cursor blink for text inputs.
func (m Model) Init() tea.Cmd { return textinput.Blink }
```

func Run(subs []string, target string, workers int, timeout time.Duration) error {
	m := NewModel(subs, target, workers, timeout)
	prog := tea.NewProgram(m)
	go func() {
		// Wait for StartTestsMsg-driven runs: the model emits StartTestsMsg
		// on Enter; the test goroutine is started from the Branch below via
		// startBackgroundRun. Kept explicit so the flow stays reviewable.
	}()
	_, err := prog.Run()
	return err
}

// startBackgroundRun executes fetch -> dedupe -> RunPool off the UI thread
// and forwards every result with prog.Send (goroutine-safe in BubbleTea).
func startBackgroundRun(prog *tea.Program, cfg FormConfig) {
	go func() {
		start := time.Now()
		ctx, cancel := context.WithCancel(context.Background())
		defer cancel()
		client := &http.Client{Timeout: 30 * time.Second}
		fctx, fcancel := context.WithTimeout(ctx, 60*time.Second)
		rawLinks := subscription.FetchSubscriptions(fctx, cfg.Subs, client)
		fcancel()
		configs := parser.DeduplicateConfigs(rawLinks)
		col := &Collector{}
		if len(configs) == 0 {
			prog.Send(DoneMsg{})
			return
		}
		tester.RunPool(ctx, configs, cfg.TargetURL, cfg.Workers, cfg.Timeout, func(res types.TestResult) {
			col.OnResult(res)
			prog.Send(ResultMsg{Res: res})
		})
		tested, okCount, failCount, links := col.Snapshot()
		saved := ""
		if len(links) > 0 {
			_ = os.MkdirAll("results", 0755)

		}
		prog.Send(DoneMsg{Tested: tested, Success: okCount, Failed: failCount, Links: links, SavedFile: saved, Elapsed: time.Since(start)})
	}()
}
